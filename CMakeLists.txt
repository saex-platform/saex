cmake_minimum_required(VERSION 3.25)
project(SAEX VERSION 0.1.11 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(CTest)
find_package(Threads REQUIRED)

add_library(saex_build_options INTERFACE)
if(MSVC)
  target_compile_options(saex_build_options INTERFACE /W4 /WX /permissive- /utf-8 /EHsc)
else()
  target_compile_options(saex_build_options INTERFACE -Wall -Wextra -Wpedantic -Werror)
endif()

add_library(saex_core STATIC
  src/core/control_clock.cpp
  src/core/lease_authority.cpp
  src/core/bounded_inbox.cpp
  src/contracts/fixture_codec.cpp)
target_include_directories(saex_core PUBLIC include)
target_link_libraries(saex_core PUBLIC saex_build_options Threads::Threads)

add_executable(saex_contract_probe src/tools/contract_probe.cpp)
target_link_libraries(saex_contract_probe PRIVATE saex_core)

# Portable observation metadata; has no Windows process or Plugin-SDK dependency.
add_library(saex_loader_lifecycle STATIC src/engine/loader/loader_mapping_ledger.cpp)
target_include_directories(saex_loader_lifecycle PUBLIC include)
target_link_libraries(saex_loader_lifecycle PUBLIC saex_build_options)

# Preflight must compile without or before any Plugin-SDK integration.
add_library(saex_engine_preflight STATIC
  src/engine/bootstrap/pe_image.cpp
  src/engine/bootstrap/profile_gate.cpp
  src/engine/bootstrap/bootstrap_session.cpp)
target_include_directories(saex_engine_preflight PUBLIC include)
target_link_libraries(saex_engine_preflight PUBLIC saex_build_options)
if(WIN32)
  target_sources(saex_engine_preflight PRIVATE src/engine/bootstrap/windows_image_reader.cpp
    src/engine/bootstrap/windows_file_observation.cpp)
  target_link_libraries(saex_engine_preflight PRIVATE bcrypt)
  target_compile_definitions(saex_engine_preflight PRIVATE _WIN32_WINNT=0x0A00)
  add_executable(saex_engine_image_probe src/engine/bootstrap/image_probe.cpp)
  target_compile_definitions(saex_engine_image_probe PRIVATE _WIN32_WINNT=0x0A00)
  target_link_libraries(saex_engine_image_probe PRIVATE saex_engine_preflight bcrypt)
  # Development observer runs outside the game; never link it into the bootstrap DLL.
  add_library(saex_engine_observer STATIC src/engine/observer/suspended_image.cpp src/engine/observer/launch_context.cpp)
  target_link_libraries(saex_engine_observer PUBLIC saex_engine_preflight)
  target_compile_definitions(saex_engine_observer PRIVATE _WIN32_WINNT=0x0A00)
  add_executable(saex_engine_process_probe src/engine/observer/process_probe.cpp)
  target_link_libraries(saex_engine_process_probe PRIVATE saex_engine_observer)
  if(CMAKE_SIZEOF_VOID_P EQUAL 4)
    add_library(saex_loader_observer STATIC src/engine/loader/loader_observation.cpp src/engine/loader/loader_policy.cpp)
    target_link_libraries(saex_loader_observer PUBLIC saex_engine_observer saex_loader_lifecycle PRIVATE bcrypt)
    target_compile_definitions(saex_loader_observer PRIVATE _WIN32_WINNT=0x0A00)
    add_executable(saex_engine_loader_probe src/engine/loader/loader_probe.cpp)
    target_link_libraries(saex_engine_loader_probe PRIVATE saex_loader_observer)
    add_library(saex_bootstrap SHARED src/engine/bootstrap/bootstrap_module.cpp
      src/engine/bootstrap/bootstrap_module.def)
    target_link_libraries(saex_bootstrap PRIVATE saex_engine_preflight)
    target_compile_definitions(saex_bootstrap PRIVATE _WIN32_WINNT=0x0A00)
    if(MSVC)
      target_link_options(saex_bootstrap PRIVATE /INCREMENTAL:NO
        "/MAP:$<TARGET_FILE_DIR:saex_bootstrap>/saex_bootstrap.map")
    endif()
  endif()
endif()

if(BUILD_TESTING)
  add_executable(saex_loader_mapping_tests tests/engine/loader_mapping_tests.cpp)
  target_link_libraries(saex_loader_mapping_tests PRIVATE saex_loader_lifecycle)
  add_test(NAME native.loader_mappings COMMAND saex_loader_mapping_tests)
  set_tests_properties(native.loader_mappings PROPERTIES TIMEOUT 30)
  add_executable(saex_core_tests tests/native/core_tests.cpp)
  target_link_libraries(saex_core_tests PRIVATE saex_core)
  add_test(NAME native.foundation COMMAND saex_core_tests)
  set_tests_properties(native.foundation PROPERTIES TIMEOUT 30)
  add_executable(saex_engine_preflight_tests tests/engine/preflight_tests.cpp)
  target_link_libraries(saex_engine_preflight_tests PRIVATE saex_engine_preflight)
  add_test(NAME native.engine_preflight COMMAND saex_engine_preflight_tests)
  set_tests_properties(native.engine_preflight PROPERTIES TIMEOUT 30)
  add_executable(saex_bootstrap_session_tests tests/engine/bootstrap_session_tests.cpp)
  target_link_libraries(saex_bootstrap_session_tests PRIVATE saex_engine_preflight Threads::Threads)
  add_test(NAME native.bootstrap_session COMMAND saex_bootstrap_session_tests)
  set_tests_properties(native.bootstrap_session PROPERTIES TIMEOUT 30)
  if(WIN32)
    add_executable(saex_launch_context_fixture tests/engine/launch_context_fixture.cpp)
    target_link_libraries(saex_launch_context_fixture PRIVATE saex_build_options)
    add_executable(saex_launch_context_tests tests/engine/launch_context_tests.cpp)
    target_link_libraries(saex_launch_context_tests PRIVATE saex_engine_observer)
    target_compile_definitions(saex_launch_context_tests PRIVATE _WIN32_WINNT=0x0A00)
    add_dependencies(saex_launch_context_tests saex_launch_context_fixture)
    add_test(NAME native.launch_context COMMAND saex_launch_context_tests $<TARGET_FILE:saex_launch_context_fixture>)
    set_tests_properties(native.launch_context PROPERTIES TIMEOUT 60)
    add_executable(saex_suspended_fixture tests/engine/suspended_fixture.cpp)
    target_link_libraries(saex_suspended_fixture PRIVATE saex_build_options)
    set_target_properties(saex_suspended_fixture PROPERTIES OUTPUT_NAME "saex observer fixture")
    add_executable(saex_suspended_image_tests tests/engine/suspended_image_tests.cpp)
    target_link_libraries(saex_suspended_image_tests PRIVATE saex_engine_observer Threads::Threads)
    target_compile_definitions(saex_suspended_image_tests PRIVATE _WIN32_WINNT=0x0A00)
    add_dependencies(saex_suspended_image_tests saex_suspended_fixture)
    add_test(NAME native.suspended_image COMMAND saex_suspended_image_tests $<TARGET_FILE:saex_suspended_fixture>)
    set_tests_properties(native.suspended_image PROPERTIES TIMEOUT 60)
  endif()
  if(TARGET saex_bootstrap)
    add_library(saex_loader_fixture_dll SHARED tests/engine/loader_fixture_dll.cpp)
    add_executable(saex_loader_fixture tests/engine/loader_fixture.cpp)
    target_link_libraries(saex_loader_fixture PRIVATE saex_loader_fixture_dll saex_build_options)
    target_link_libraries(saex_loader_fixture_dll PRIVATE saex_build_options)
    # Only the owned loader canary fixtures use static CRT, keeping the test closure small.
    set_property(TARGET saex_loader_fixture saex_loader_fixture_dll PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    add_executable(saex_loader_observation_tests tests/engine/loader_observation_tests.cpp)
    target_link_libraries(saex_loader_observation_tests PRIVATE saex_loader_observer Threads::Threads)
    target_compile_definitions(saex_loader_observation_tests PRIVATE _WIN32_WINNT=0x0A00)
    add_dependencies(saex_loader_observation_tests saex_loader_fixture)
    add_test(NAME native.loader_observation COMMAND saex_loader_observation_tests $<TARGET_FILE:saex_loader_fixture> $<TARGET_FILE:saex_loader_fixture_dll>)
    set_tests_properties(native.loader_observation PROPERTIES TIMEOUT 60)
    foreach(entry_mode IN ITEMS mutate fault stall)
      add_executable(saex_entry_fixture_${entry_mode} tests/engine/entry_fixture.cpp)
      target_compile_definitions(saex_entry_fixture_${entry_mode} PRIVATE SAEX_ENTRY_${entry_mode}=1)
      target_link_libraries(saex_entry_fixture_${entry_mode} PRIVATE saex_loader_fixture_dll saex_build_options)
      set_property(TARGET saex_entry_fixture_${entry_mode} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endforeach()
    add_executable(saex_entry_observation_tests tests/engine/entry_observation_tests.cpp)
    target_link_libraries(saex_entry_observation_tests PRIVATE saex_loader_observer Threads::Threads)
    target_compile_definitions(saex_entry_observation_tests PRIVATE _WIN32_WINNT=0x0A00)
    add_dependencies(saex_entry_observation_tests saex_loader_fixture saex_entry_fixture_mutate saex_entry_fixture_fault saex_entry_fixture_stall)
    add_test(NAME native.entry_observation COMMAND saex_entry_observation_tests
      $<TARGET_FILE:saex_loader_fixture> $<TARGET_FILE:saex_loader_fixture_dll>
      $<TARGET_FILE:saex_entry_fixture_mutate> $<TARGET_FILE:saex_entry_fixture_fault> $<TARGET_FILE:saex_entry_fixture_stall>)
    set_tests_properties(native.entry_observation PROPERTIES TIMEOUT 60)
    add_executable(saex_bootstrap_module_tests tests/engine/bootstrap_module_tests.cpp)
    target_link_libraries(saex_bootstrap_module_tests PRIVATE saex_build_options)
    target_include_directories(saex_bootstrap_module_tests PRIVATE include)
    target_compile_definitions(saex_bootstrap_module_tests PRIVATE _WIN32_WINNT=0x0A00)
    add_dependencies(saex_bootstrap_module_tests saex_bootstrap)
    add_test(NAME native.bootstrap_module COMMAND saex_bootstrap_module_tests $<TARGET_FILE:saex_bootstrap>)
    set_tests_properties(native.bootstrap_module PROPERTIES TIMEOUT 30)
  endif()
endif()
